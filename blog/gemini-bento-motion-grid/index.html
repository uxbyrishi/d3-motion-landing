Under the Hood: The Motion Math Language

To eliminate the jerkiness typical of standard CSS cubic-beziers, the motion engine relies on three core mathematical helpers implemented directly in vanilla JavaScript:

motion-helpers.js — Damped Spring & Odometer Physics 
// 1. Damped Spring From Rest (zeta = 0.55, ~12% overshoot, zero start jerk)
function spring(p) {
  p = Math.max(0, Math.min(1, p));
  const z = 0.55, w = 9.2;
  const decay = Math.exp(-z * w * p);
  const osc = Math.cos(w * Math.sqrt(1 - z * z) * p);
  return 1 - decay * osc;
}

// 2. Continuous State Wraps (Prevents wrapping snaps)
function st(t, P, o) {
  const rel = (t - o) % P;
  const l = rel < 0 ? rel + P : rel;
  return { k: Math.floor((t - o) / P), l, pre: t < o };
}

// 3. Tabular Odometer Digit Rolling
function roll(from, to, progress) {
  const p = spring(progress);
  return `<span class="tnum" style="transform: translateY(${(to - from) * (1 - p)}px)">${to}</span>`;
}
